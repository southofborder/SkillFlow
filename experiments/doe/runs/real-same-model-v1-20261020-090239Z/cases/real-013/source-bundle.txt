57: function imageToBase64(imagePath) {
58:   const absolutePath = path.resolve(imagePath);
59: 
60:   // Validate file extension is an allowed image type
61:   const ext = path.extname(absolutePath).toLowerCase();
62:   const mimeTypes = {
63:     '.jpg': 'image/jpeg',
64:     '.jpeg': 'image/jpeg',
65:     '.png': 'image/png',
66:     '.gif': 'image/gif',
67:     '.webp': 'image/webp',
68:   };
69: 
70:   if (!mimeTypes[ext]) {
71:     throw new Error(`Unsupported image format: ${ext}. Allowed: ${Object.keys(mimeTypes).join(', ')}`);
72:   }
73: 
74:   if (!fs.existsSync(absolutePath)) {
75:     throw new Error(`Image not found: ${absolutePath}`);
76:   }
77: 
78:   const imageData = fs.readFileSync(absolutePath);
79:   const base64 = imageData.toString('base64');
80:   const mimeType = mimeTypes[ext];
81: 
82:   return `data:${mimeType};base64,${base64}`;
83: }
