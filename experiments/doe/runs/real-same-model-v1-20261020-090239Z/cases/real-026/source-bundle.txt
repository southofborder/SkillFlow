FILE bilibili_api.py
1: #!/usr/bin/env python3
2: """
3: B站 API 工具集
4: 
5: 包含：
6: - WBI 签名生成
7: - 热门视频获取
8: - AI 视频总结获取
9: - 视频详情获取
10: 
11: 参考文档：
12: - https://socialsisteryi.github.io/bilibili-API-collect/docs/misc/sign/wbi.html
13: - https://socialsisteryi.github.io/bilibili-API-collect/docs/video/summary.html
14: """
15: 
16: import hashlib
17: import json
18: import time
19: import urllib.parse
20: from functools import reduce
21: from typing import Any
22: 
23: import requests
24: 
25: 
26: # WBI 签名用的混淆表
27: MIXIN_KEY_ENC_TAB = [
28:     46, 47, 18, 2, 53, 8, 23, 32, 15, 50, 10, 31, 58, 3, 45, 35,
29:     27, 43, 5, 49, 33, 9, 42, 19, 29, 28, 14, 39, 12, 38, 41, 13,
30:     37, 48, 7, 16, 24, 55, 40, 61, 26, 17, 0, 1, 60, 51, 30, 4,
31:     22, 25, 54, 21, 56, 59, 6, 63, 57, 62, 11, 36, 20, 34, 44, 52
32: ]
33: 
34: 
35: class BilibiliAPI:
36:     """B站 API 客户端"""
37: 
38:     def __init__(self, sessdata: str = "", bili_jct: str = "", buvid3: str = "", dedeuserid: str = "", all_cookies: dict = None):
39:         """
40:         初始化 API 客户端
41: 
42:         Args:
43:             sessdata: SESSDATA cookie
44:             bili_jct: bili_jct cookie
45:             buvid3: buvid3 cookie
46:             dedeuserid: DedeUserID cookie
47:             all_cookies: 所有 cookies 字典（可选，直接设置全部 cookies）
48:         """
49:         self.session = requests.Session()
50:         self.session.headers.update({
51:             "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36",
52:             "Referer": "https://www.bilibili.com",
53:         })
54: 
55:         # 如果提供了全部 cookies，直接设置
56:         if all_cookies:
57:             for key, value in all_cookies.items():
58:                 self.session.cookies.set(key, value, domain=".bilibili.com")
59:         else:
60:             # 设置单独的 cookies
61:             if sessdata:
62:                 self.session.cookies.set("SESSDATA", sessdata, domain=".bilibili.com")
63:             if bili_jct:
64:                 self.session.cookies.set("bili_jct", bili_jct, domain=".bilibili.com")
65:             if buvid3:
66:                 self.session.cookies.set("buvid3", buvid3, domain=".bilibili.com")
67:             if dedeuserid:
68:                 self.session.cookies.set("DedeUserID", dedeuserid, domain=".bilibili.com")
69: 
70:         # WBI 密钥缓存
71:         self._img_key = ""
72:         self._sub_key = ""
73: 
74:     def _get_mixin_key(self, orig: str) -> str:
75:         """生成混淆后的密钥"""
76:         return reduce(lambda s, i: s + orig[i], MIXIN_KEY_ENC_TAB, '')[:32]
77: 
78:     def _get_wbi_keys(self) -> tuple[str, str]:
79:         """获取 WBI 签名所需的 img_key 和 sub_key"""
80:         if self._img_key and self._sub_key:
81:             return self._img_key, self._sub_key
82: 
83:         resp = self.session.get("https://api.bilibili.com/x/web-interface/nav")
84:         data = resp.json()
85: 
86:         if data["code"] != 0:
87:             raise Exception(f"获取 WBI 密钥失败: {data['message']}")
88: 
89:         img_url = data["data"]["wbi_img"]["img_url"]
90:         sub_url = data["data"]["wbi_img"]["sub_url"]
91: 
92:         # 从 URL 中提取密钥
93:         self._img_key = img_url.rsplit('/', 1)[1].split('.')[0]
94:         self._sub_key = sub_url.rsplit('/', 1)[1].split('.')[0]
95: 
96:         return self._img_key, self._sub_key
97: 
98:     def _sign_params(self, params: dict[str, Any]) -> dict[str, Any]:
99:         """对请求参数进行 WBI 签名"""
100:         img_key, sub_key = self._get_wbi_keys()
101:         mixin_key = self._get_mixin_key(img_key + sub_key)
102: 
103:         # 添加时间戳
104:         params["wts"] = int(time.time())
105: 
106:         # 按 key 排序
107:         params = dict(sorted(params.items()))
108: 
109:         # 过滤特殊字符并 URL 编码
110:         query = urllib.parse.urlencode(params)
111: 
112:         # 计算签名
113:         w_rid = hashlib.md5((query + mixin_key).encode()).hexdigest()
114:         params["w_rid"] = w_rid
115: 
116:         return params
117: 
118:     def get_popular_videos(self, page: int = 1, page_size: int = 20) -> list[dict]:
119:         """
120:         获取热门视频列表
121:

31: def call_openrouter(api_key: str, model: str, prompt: str, max_tokens: int = 500, max_retries: int = 3) -> str:
32:     """
33:     调用 OpenRouter API 生成内容（带重试机制）
34: 
35:     Args:
36:         api_key: OpenRouter API Key
37:         model: 模型名称，如 "anthropic/claude-sonnet-4.5"
38:         prompt: 提示词
39:         max_tokens: 最大生成 token 数
40:         max_retries: 最大重试次数
41: 
42:     Returns:
43:         生成的文本
44:     """
45:     for attempt in range(max_retries):
46:         try:
47:             response = requests.post(
48:                 "https://openrouter.ai/api/v1/chat/completions",
49:                 headers={
50:                     "Authorization": f"Bearer {api_key}",
51:                     "Content-Type": "application/json",
52:                 },
53:                 json={
54:                     "model": model,
55:                     "messages": [{"role": "user", "content": prompt}],
56:                     "max_tokens": max_tokens,
57:                     # 禁用所有模型的 extended thinking/reasoning 模式
58:                     # effort: "none" 完全禁用推理模式，避免 JSON 输出被截断或格式错乱
59:                     # 适用于 Claude、Gemini、DeepSeek 等支持 reasoning 的模型
60:                     "reasoning": {
61:                         "effort": "none"
62:                     },
63:                 },
64:                 timeout=60,
65:             )
66: 
67:             if response.status_code != 200:
68:                 print(f"  [WARNING] OpenRouter API 错误: {response.status_code}")
69:                 if attempt < max_retries - 1:
70:                     time.sleep(2)  # 等待2秒后重试
71:                     continue
72:                 return ""
73: 
74:             data = response.json()
75:             return data["choices"][0]["message"]["content"].strip()
76: 
77:         except (requests.exceptions.SSLError, requests.exceptions.ConnectionError) as e:
78:             # SSL 或连接错误，等待后重试
79:             if attempt < max_retries - 1:
80:                 print(f"  [RETRY] 网络错误，{2*(attempt+1)}秒后重试 ({attempt+1}/{max_retries})...")
81:                 time.sleep(2 * (attempt + 1))  # 递增等待时间
82:                 continue
83:             print(f"  [WARNING] OpenRouter 调用失败（已重试{max_retries}次）: {e}")
84:             return ""
85: 
86:         except Exception as e:
87:             print(f"  [WARNING] OpenRouter 调用失败: {e}")
88:             return ""
89: 
90:     return ""
