2: # Canva CLI Helper
3: # Usage: canva.sh <command> [args]
4:

6:

13:     fi
14:     jq -r '.access_token' "$TOKEN_FILE"
15: }
16: 
17: api() {
18:     local method="$1"
19:     local endpoint="$2"
20:     local data="$3"
21:     local token=$(get_token)
22:     
23:     if [ -n "$data" ]; then
24:         curl -s -X "$method" "https://api.canva.com/rest/v1$endpoint" \
25:             -H "Authorization: Bearer $token" \
26:             -H "Content-Type: application/json" \
27:             -d "$data"
28:     else
29:         curl -s -X "$method" "https://api.canva.com/rest/v1$endpoint" \
30:             -H "Authorization: Bearer $token"
31:     fi
32: }
33: 
34: case "$1" in
35:     designs|list)
36:         echo "📋 Listing designs..."
37:         api GET "/designs" | jq '.items[] | {id, name: .title, created: .created_at}'
38:         ;;
39:     
40:     get)
41:         if [ -z "$2" ]; then
42:             echo "Usage: canva.sh get <design_id>"
43:             exit 1
44:         fi
45:         api GET "/designs/$2" | jq .
46:         ;;
47:     
48:     templates)
49:         echo "📋 Listing brand templates..."

51:         ;;
52:     
53:     export)
54:         if [ -z "$2" ]; then
55:             echo "Usage: canva.sh export <design_id> [format: png|jpg|pdf]"
56:             exit 1
57:         fi

77:                 echo "$STATUS" | jq .
78:                 exit 1

81:         done
