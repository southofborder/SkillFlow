12: function _loadToken() {
13:   if (!fs.existsSync(TOKEN_FILE)) return null;
14: 
15:   try {
16:     const data = JSON.parse(fs.readFileSync(TOKEN_FILE, 'utf-8'));
17:     const expiresAtStr = data.expires_at;
18:     if (!expiresAtStr) return null;
19: 
20:     const expiresAt = new Date(expiresAtStr);
21:     if (Date.now() >= expiresAt.getTime()) return null;
22: 
23:     return data.token || null;
24:   } catch (e) {
25:     console.error(`Warning: Failed to load token file: ${e.message}`);
26:     return null;
27:   }

30: function _httpRequest(reqUrl, headers = {}) {
31:   return new Promise((resolve, reject) => {
32:     const parsed = new URL(reqUrl);
33:     const mod = parsed.protocol === 'https:' ? https : http;
34:     const options = {
35:       hostname: parsed.hostname,
36:       port: parsed.port,
37:       path: parsed.pathname + parsed.search,
38:       method: 'GET',
39:       headers: { 'User-Agent': USER_AGENT, ...headers },
40:     };
41: 
42:     const req = mod.request(options, (res) => {
43:       let body = '';
44:       res.on('data', (chunk) => (body += chunk));
45:       res.on('end', () => resolve({ status: res.statusCode, body }));
46:     });
47:     req.on('error', reject);
48:     req.end();
49:   });
