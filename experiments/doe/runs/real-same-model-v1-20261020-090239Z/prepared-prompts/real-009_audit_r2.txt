FILE scripts/moltbook.sh
1: #!/usr/bin/env bash
2: # Moltbook CLI helper
3: 
4: CONFIG_FILE="${HOME}/.config/moltbook/credentials.json"
5: OPENCLAW_AUTH="${HOME}/.openclaw/auth-profiles.json"
6: API_BASE="https://www.moltbook.com/api/v1"
7: 
8: # Load API key - check OpenClaw auth first, then fallback to credentials file
9: API_KEY=""
10: 
11: # Try OpenClaw auth system first
12: if [[ -f "$OPENCLAW_AUTH" ]]; then
13:     if command -v jq &> /dev/null; then
14:         API_KEY=$(jq -r '.moltbook.api_key // empty' "$OPENCLAW_AUTH" 2>/dev/null)
15:     fi
16: fi
17: 
18: # Fallback to credentials file
19: if [[ -z "$API_KEY" && -f "$CONFIG_FILE" ]]; then
20:     if command -v jq &> /dev/null; then
21:         API_KEY=$(jq -r .api_key "$CONFIG_FILE")
22:     else
23:         # Fallback: extract key with grep/sed
24:         API_KEY=$(grep '"api_key"' "$CONFIG_FILE" | sed 's/.*"api_key"[[:space:]]*:[[:space:]]*"\([^"]*\)".*/\1/')
25:     fi
26: fi
27: 
28: if [[ -z "$API_KEY" || "$API_KEY" == "null" ]]; then
29:     echo "Error: Moltbook credentials not found"
30:     echo ""
31:     echo "Option 1 - OpenClaw auth (recommended):"
32:     echo "  openclaw agents auth add moltbook --token your_api_key"
33:     echo ""
34:     echo "Option 2 - Credentials file:"
35:     echo "  mkdir -p ~/.config/moltbook"
36:     echo "  echo '{\"api_key\":\"your_key\",\"agent_name\":\"YourName\"}' > ~/.config/moltbook/credentials.json"
37:     exit 1
38: fi
39: 
40: # Helper function for API calls
41: api_call() {
42:     local method=$1
43:     local endpoint=$2
44:     local data=$3
45:     
46:     if [[ -n "$data" ]]; then
47:         curl -s -X "$method" "${API_BASE}${endpoint}" \
48:             -H "Authorization: Bearer ${API_KEY}" \
49:             -H "Content-Type: application/json" \
50:             -d "$data"
51:     else
52:         curl -s -X "$method" "${API_BASE}${endpoint}" \
53:             -H "Authorization: Bearer ${API_KEY}" \
54:             -H "Content-Type: application/json"
55:     fi
56: }
57: 
58: # Parse JSON helper (works without jq)
59: parse_json() {
60:     local json="$1"
61:     local key="$2"
62:     if command -v jq &> /dev/null; then
63:         echo "$json" | jq -r "$key"
64:     else
65:         # Simple fallback for basic extraction
66:         echo "$json" | grep -o "\"$key\":\"[^\"]*\"" | head -1 | cut -d'"' -f4
67:     fi
68: }
69: 
70: # Commands
71: case "${1:-}" in
72:     hot)
73:         limit="${2:-10}"
74:         echo "Fetching hot posts..."
75:         api_call GET "/posts?sort=hot&limit=${limit}"
76:         ;;
77:     new)
78:         limit="${2:-10}"
79:         echo "Fetching new posts..."
80:         api_call GET "/posts?sort=new&limit=${limit}"
81:         ;;
82:     post)
83:         post_id="$2"
84:         if [[ -z "$post_id" ]]; then
85:             echo "Usage: moltbook post POST_ID"
86:             exit 1
87:         fi
88:         api_call GET "/posts/${post_id}"
89:         ;;
90:     reply)
91:         post_id="$2"
92:         content="$3"
93:         if [[ -z "$post_id" || -z "$content" ]]; then
