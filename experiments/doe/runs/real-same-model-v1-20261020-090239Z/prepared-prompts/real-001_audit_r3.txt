90:     """POST JSON and return parsed response. Retries on transient errors."""
91:     data = json.dumps(payload).encode("utf-8")
92:     for attempt in range(1, retries + 1):
93:         try:
94:             req = Request(url, data=data, headers=headers, method="POST")
95:             with urlopen(req, timeout=30) as resp:
96:                 return json.loads(resp.read().decode("utf-8"))

98:             if attempt == retries:
