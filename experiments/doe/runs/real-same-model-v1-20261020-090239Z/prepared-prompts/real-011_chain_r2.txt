1: #!/usr/bin/env python3
2: """
3: Local PPTX operation tool (Layer 3, no backend required).
4: Dependency: pip install python-pptx
5: 
6: Subcommands:
7:   info     View basic file info (slide count, slide titles)
8:   delete   Delete specified slides (1-based, supports multiple and ranges, e.g. 3,5,7-9)
9:   reorder  Reorder slide sequence (e.g. 2,1,4,3,5)
10:   extract  Extract selected slides into a new file
11:   merge    Merge multiple pptx files
12: 
13: Usage examples:
14:   python scripts/local_pptx_ops.py info --file my.pptx
15:   python scripts/local_pptx_ops.py delete --file my.pptx --slides 3,5,7-9
16:   python scripts/local_pptx_ops.py delete --file my.pptx --slides 3,5,7-9 -o trimmed.pptx
17:   python scripts/local_pptx_ops.py reorder --file my.pptx --order 2,1,4,3,5
18:   python scripts/local_pptx_ops.py extract --file my.pptx --slides 1-3 -o subset.pptx
19:   python scripts/local_pptx_ops.py merge --files a.pptx b.pptx -o merged.pptx
20: """
21: 
22: import argparse
23: import os
24: import sys
25: import shutil
26: import copy
27: import warnings
28: 
29: try:
30:     from pptx import Presentation
31:     from pptx.util import Inches, Pt
32:     from lxml import etree
33: except ImportError:
34:     print("Missing dependency, please run: pip install python-pptx", file=sys.stderr)
35:     sys.exit(1)
36: 
37: 
38: # ---------------------------------------------------------------------------
39: # Parse slide number string (1-based), return 0-based index list
40: # ---------------------------------------------------------------------------
41: 
42: def parse_slide_spec(spec: str, total: int) -> list[int]:
43:     """Parse a slide number string like '1,3,5-8,10', return a deduplicated sorted list of 0-based indices."""
44:     indices = set()
45:     for part in spec.split(","):
46:         part = part.strip()
47:         if not part:
48:             continue
49:         if "-" in part:
50:             lo, hi = part.split("-", 1)
51:             lo_i, hi_i = int(lo.strip()), int(hi.strip())
52:             if lo_i < 1 or hi_i > total:
53:                 raise ValueError(f"Slide range {lo_i}-{hi_i} is out of bounds for a file with {total} slides")
54:             for i in range(lo_i - 1, hi_i):
55:                 indices.add(i)
56:         else:
57:             n = int(part)
58:             if n < 1 or n > total:
59:                 raise ValueError(f"Slide number {n} is out of bounds for a file with {total} slides")
60:             indices.add(n - 1)
61:     return sorted(indices)
62: 
63: 
64: def parse_order_spec(spec: str, total: int) -> list[int]:
65:     """Parse an order string for reorder, preserving user-specified order, return 0-based index list."""
66:     result = []
67:     seen = set()
68:     for part in spec.split(","):
69:         part = part.strip()
70:         if not part:
71:             continue
72:         n = int(part)
73:         if n < 1 or n > total:
74:             raise ValueError(f"Slide number {n} is out of bounds for a file with {total} slides")
75:         idx = n - 1
76:         if idx in seen:
77:             raise ValueError(f"Slide number {n} appears more than once")
78:         seen.add(idx)
79:         result.append(idx)
80:     return result
81: 
82: 
83: # ---------------------------------------------------------------------------
84: # Core operation: remove a single slide (XML manipulation, not natively supported by python-pptx)
85: # ---------------------------------------------------------------------------
86: 
87: def _remove_slide(prs: "Presentation", index: int) -> None:
88:     """Remove the slide at the given index (0-based) from the Presentation."""
89:     xml_slides = prs.slides._sldIdLst
90:     slide_elem = xml_slides[index]
91:     rId = slide_elem.get("{http://schemas.openxmlformats.org/officeDocument/2006/relationships}id")
92:     xml_slides.remove(slide_elem)
93:     # Remove the corresponding relationship from rels to avoid "Duplicate name" warnings on save
94:     prs.part.rels.pop(rId)
95: 
96: 
97: # ---------------------------------------------------------------------------
98: # Subcommand: info
99: # ---------------------------------------------------------------------------
100: 
101: def cmd_info(args):
102:     prs = Presentation(args.file)
103:     total = len(prs.slides)

30: def build_multipart_body(file_path: str):
31:     """Build a multipart/form-data request body."""
32:     boundary = f"----FormBoundary{uuid.uuid4().hex[:16]}"
33:     body = io.BytesIO()
34:     file_name = os.path.basename(file_path)
35:     mime_type = mimetypes.guess_type(file_path)[0] or "application/octet-stream"
36: 
37:     body.write(f"--{boundary}\r\n".encode())
38:     body.write(f'Content-Disposition: form-data; name="file"; filename="{file_name}"\r\n'.encode())
39:     body.write(f"Content-Type: {mime_type}\r\n\r\n".encode())
40:     with open(file_path, "rb") as f:
41:         body.write(f.read())
42:     body.write(b"\r\n")
43:     body.write(f"--{boundary}--\r\n".encode())

27: def build_multipart(fields: dict, files: dict) -> tuple[bytes, str]:
28:     """Build a multipart/form-data request body, returns (body_bytes, content_type).
29: 
30:     fields: {"field_name": "value", ...}
31:     files:  {"field_name": (filename, file_bytes, mime_type), ...}
32:     """
33:     boundary = uuid.uuid4().hex
34:     parts = []
35: 
36:     for name, value in fields.items():
37:         parts.append(
38:             f'--{boundary}\r\n'
39:             f'Content-Disposition: form-data; name="{name}"\r\n\r\n'
40:             f'{value}\r\n'
41:         )

45:             f'--{boundary}\r\n'
46:             f'Content-Disposition: form-data; name="{name}"; filename="{filename}"\r\n'
47:             f'Content-Type: {mime}\r\n\r\n'
48:         )

51:         parts_bytes = b''.join(
52:             p.encode() if isinstance(p, str) else p for p in parts
53:         ) + data + b'\r\n'
54:         parts = [parts_bytes]

2: """
3: web_search.py - Call local web search API and print results.
4:

7: """

16: from constant import SKYWORK_GATEWAY_URL, POD_TYPE

18: 
19: def search(query: str, api_key: str) -> str:
20:     """Call web_search API and return formatted text of results."""
21:     url = f"{SKYWORK_GATEWAY_URL}/web_search"
22:     payload = {"query": query, "source_platform": "skyclaw" if POD_TYPE == "skyclaw" else ""}
23:     body = json.dumps(payload).encode("utf-8")
24:     headers = {"Content-Type": "application/json"}
25:     if api_key:
26:         headers["Authorization"] = f"Bearer {api_key}"
27:     req = urllib.request.Request(
28:         url,
29:         data=body,
30:         method="POST",
31:         headers=headers,
32:     )
33:     try:
34:         with urllib.request.urlopen(req, timeout=30) as resp:
35:             raw = resp.read().decode("utf-8")
36:     except urllib.error.URLError as e:
37:         print(f"[ERROR] Error call web_search, try only 1 time more: {e}", file=sys.stderr)
38:         sys.exit(1)
39: 
40:     try:
41:         data = json.loads(raw)
42:     except json.JSONDecodeError:
43:         return raw  # fallback: return raw if not JSON
44: 
45:     results = data.get("search_res", [])
46:     if not results:
47:         return "(no results)"
48: 
49:     lines = []
50:     for i, item in enumerate(results, 1):
51:         content = (item.get("content") or "").strip()
52:         source_url = item.get("url", "")
53:         lines.append(f"[result-{i}] {source_url}\n{content}")
54:     return "\n\n".join(lines)
55: 
56: 
57: def main():
58:     parser = argparse.ArgumentParser(description="Call local web_search API")
59:     parser.add_argument("queries", nargs="+", help="One or more search queries (max 3)")
60:     args = parser.parse_args()
61:

127: ## FAQ
128: 
129: **Q: Do page numbers change after deletion?**
130: A: Yes. After deletion, page numbers are renumbered starting from 1. If you need to delete multiple slides, it's recommended to pass them all at once (`--slides 3,5,7`) rather than running multiple times.
131: 
132: **Q: What if styles are wrong after merging?**
133: A: Merging uses deep-copy of the shape tree; themes/masters come from the first file. If there are significant style differences, manual adjustments in PowerPoint are recommended.
134: 
135: **Q: Does python-pptx support deleting slides?**
136: A: Not natively. This tool operates on the underlying XML (`_sldIdLst`), consistent with common python-pptx community approaches.
