2: # -*- coding: utf-8 -*-
3: """
4: Wait for Window - Wait until a window with specific title exists
5: Usage: py wait_for_window.py "Window Title" [timeout_seconds]
6: """
7: import sys
8: import io
9: import time
10: from pywinauto import Desktop
11: 
12: sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding='utf-8', errors='replace')
13: 
14: if len(sys.argv) < 2:
15:     print("Usage: py wait_for_window.py \"Window Title\" [timeout]")
16:     sys.exit(1)
17: 
18: window_title = sys.argv[1]
19: timeout = int(sys.argv[2]) if len(sys.argv) > 2 else 30
20: 
21: try:
22:     desktop = Desktop(backend="uia")
23:     start_time = time.time()
24:     
25:     while time.time() - start_time < timeout:
26:         windows = desktop.windows()
27:         for window in windows:
28:             if window_title.lower() in window.window_text().lower():
29:                 print(f"Found window '{window.window_text()}' after {time.time() - start_time:.1f}s")
30:                 sys.exit(0)
31:         time.sleep(0.5)
32:         
33:     print(f"Timeout: Window '{window_title}' not found after {timeout}s")
34:     sys.exit(1)
35: except Exception as e:
36:     print(f"Error: {e}")
37:     sys.exit(1)
