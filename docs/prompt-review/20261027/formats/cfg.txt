Format notation: fields are required unless suffixed ?. Omission is distinct from null. ANY allows one or more branches; ONE exactly one; ALL and & require all constraints. extra:FORBIDDEN rejects unknown object keys; extra:JSON permits them. Arrays preserve order; ... describes each remaining item. Defaults apply only on omission. JSON means any JSON value. pattern is a regular-expression constraint. Named types below are definitions, not additional response fields. Application rules follow.
RESPONSE = ANY(IRAnalysisCandidate | _FailureResponse)
CandidateBlock = {"block_ref":string,"block_name":string,"data_source_kind"?:ANY(string & enum=["context","external"] | null) & default=null,"instructions"?:[...CandidateInstruction],"constraints"?:[...string],extra:FORBIDDEN} & meaning="A behavior unit proposed by the LLM.\n\nA block states only what happens inside it.  Which of its values are\nreadable elsewhere is a property of the paths through the graph, so it is\nchecked against CFG paths rather than declared here."
CandidateEdge = {"source_block_ref":string,"target_block_ref":string,"condition_text"?:ANY(string | null) & default=null,extra:FORBIDDEN} & meaning="A control-flow edge proposed by the LLM."
CandidateInstruction = {"instruction_ref":string,"opcode":string,"inputs"?:[...Operand],"outputs"?:[...Operand],"constraints"?:[...string],"metadata"?:{extra:JSON},extra:FORBIDDEN} & meaning="An instruction proposed by the LLM before canonical compilation."
IRAnalysisCandidate = {"entry_block_ref":string,"constraints"?:[...string],"declared_context_keys"?:[...string],"blocks"?:[...CandidateBlock],"edges"?:[...CandidateEdge],"diagnostics"?:[...string],extra:FORBIDDEN} & meaning="The complete untrusted analysis returned by the LLM."
Operand = {"type":OperandType,"identifier"?:ANY(string | null) & default=null,"literal_value"?:ANY(JSON | null) & default=null,"semantic_name"?:ANY(string | null) & default=null,extra:FORBIDDEN} & meaning="One typed instruction input or output.\n\nRESULT identifies an instruction result; CONTEXT_KEY identifies a declared\nruntime entry; EXTERNAL_RESOURCE identifies a file, tool, API, or service.\nOnly LITERAL stores a payload in ``literal_value``. ``semantic_name`` keeps\nthe extractor's wording when compilation assigns a result identifier.\nResult identifiers are global; a single instruction can produce several."
OperandType = string & enum=["literal","context_key","result","external_resource"] & meaning="The source or destination kind of an operand."
TaskEvidence = {"file":string & minLength=1 & pattern="\\S","start_line":integer & minimum=1,"end_line":integer & minimum=1,"reason":string & minLength=1 & pattern="\\S",extra:FORBIDDEN} & meaning="An exact physical span with an unverified relevance explanation."
_FailureResponse = {"outcome":const="cannot_assess","failure":_TaskFailure,extra:FORBIDDEN}
_TaskFailure = {"reason":string & minLength=1 & pattern="\\S","evidences":[...TaskEvidence] & minItems=1,extra:FORBIDDEN}
