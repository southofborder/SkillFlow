Format notation: fields are required unless suffixed ?. Omission is distinct from null. ANY allows one or more branches; ONE exactly one; ALL and & require all constraints. extra:FORBIDDEN rejects unknown object keys; extra:JSON permits them. Arrays preserve order; ... describes each remaining item. Defaults apply only on omission. JSON means any JSON value. pattern is a regular-expression constraint. Named types below are definitions, not additional response fields. Application rules follow.
RESPONSE = ONE(RawReviewResponse | CannotAssessResponse) & tag="outcome":{"cannot_assess":"CannotAssessResponse","completed":"RawReviewResponse"}
BlockTarget = {"block_id":string & minLength=1 & pattern="\\S","path":[...ANY(string | integer & minimum=0)],extra:FORBIDDEN}
CFGEvidence = {"target":ANY(OperandRef | InstructionTarget | BlockTarget | GraphTarget),"reason":string & minLength=1 & pattern="\\S",extra:FORBIDDEN}
CannotAssessResponse = {"outcome":const="cannot_assess","failure":ReviewFailure,extra:FORBIDDEN}
GraphTarget = {"path":[...ANY(string | integer & minimum=0)],extra:FORBIDDEN}
InstructionTarget = {"instruction_id":string & minLength=1 & pattern="\\S","path":[...ANY(string | integer & minimum=0)],extra:FORBIDDEN}
OperandRef = {"instruction_id":string & minLength=1 & pattern="\\S","side":string & enum=["input","output"],"index":integer & minimum=0,extra:FORBIDDEN}
RawDeclarationCheck = {"status":string & enum=["accepted","issue"],"reason":string & minLength=1 & pattern="\\S","finding_indexes":[...integer & minimum=0],"raw_pointer":string & minLength=1 & pattern="\\S",extra:FORBIDDEN}
RawFinding = {"domain":string & enum=["task","annotation"],"target_ids":[...string & minLength=1 & pattern="\\S"] & minItems=1,"explanation":string & minLength=1 & pattern="\\S","evidences":[...ANY(TaskEvidence | CFGEvidence | RuleEvidence)] & minItems=1,"suggestion":string & minLength=1 & pattern="\\S","task_objection"?:ANY(TaskRepairObjection | null) & default=null,extra:FORBIDDEN}
RawRequirementCheck = {"status":string & enum=["accepted","issue"],"reason":string & minLength=1 & pattern="\\S","finding_indexes":[...integer & minimum=0],"requirement_id":string & minLength=1 & pattern="\\S",extra:FORBIDDEN}
RawReviewResponse = {"outcome":const="completed","requirement_checks":[...RawRequirementCheck] & minItems=1,"declaration_checks":[...RawDeclarationCheck],"reviewed_ir_ids":[...string & minLength=1 & pattern="\\S"] & minItems=1,"findings":[...RawFinding],extra:FORBIDDEN}
ReviewFailure = {"reason":string & minLength=1 & pattern="\\S","target_ids":[...string & minLength=1 & pattern="\\S"] & minItems=1,"evidences":[...ANY(TaskEvidence | CFGEvidence | RuleEvidence)] & minItems=1,extra:FORBIDDEN}
RuleEvidence = {"rule_id":string & minLength=1 & pattern="\\S","reason":string & minLength=1 & pattern="\\S",extra:FORBIDDEN}
TaskEvidence = {"file":string & minLength=1 & pattern="\\S","start_line":integer & minimum=1,"end_line":integer & minimum=1,"reason":string & minLength=1 & pattern="\\S",extra:FORBIDDEN} & meaning="An exact physical span with an unverified relevance explanation."
TaskRepairObjection = {"code":string & enum=["missing_requirement","unsupported_requirement","broadened_requirement","condition_scope","atomicity","conflict_resolution"],"source_spans":[...TaskSourceSpan] & minItems=1,"requirement_ids":[...string & minLength=1 & pattern="\\S"],extra:FORBIDDEN} & meaning="No explanations, graph targets, demands, or arbitrary feedback context."
TaskSourceSpan = {"file":string & minLength=1 & pattern="\\S","start_line":integer & minimum=1,"end_line":integer & minimum=1,extra:FORBIDDEN}
